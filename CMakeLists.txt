cmake_minimum_required(VERSION 3.16)
project(distributed_cache LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(cache_core src/lru_cache.cpp)
target_include_directories(cache_core PUBLIC include)
add_executable(cache-server src/server.cpp)
target_link_libraries(cache-server PRIVATE cache_core Threads::Threads)
target_link_options(cache-server PRIVATE -static-libstdc++ -static-libgcc)
enable_testing()
add_executable(cache-unit-tests tests/lru_cache_test.cpp)
target_link_libraries(cache-unit-tests PRIVATE cache_core Threads::Threads)
add_test(NAME cache-unit-tests COMMAND cache-unit-tests)
